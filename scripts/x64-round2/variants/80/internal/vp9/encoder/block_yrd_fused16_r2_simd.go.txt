//go:build goexperiment.simd && (arm64 || amd64) && !purego

package encoder

import (
	"simd/archsimd"
	"unsafe"
)

// The BlockYrd scoring caller proves complete residual rows, disjoint scratch,
// and the bounded quantizer contract documented by blockYrdQuantScore.
func blockYrdHadamard16Score(input []int16, stride int, coeff []int16, r, q, d [2]int16) (int, uint32) {
	sp, cp := unsafe.Pointer(unsafe.SliceData(input)), unsafe.Pointer(unsafe.SliceData(coeff))
	hadamardBlockYrd8(sp, stride, cp)
	hadamardBlockYrd8(unsafe.Add(sp, 16), stride, unsafe.Add(cp, 128))
	hadamardBlockYrd8(unsafe.Add(sp, 16*stride), stride, unsafe.Add(cp, 256))
	hadamardBlockYrd8(unsafe.Add(sp, 16*stride+16), stride, unsafe.Add(cp, 384))

	ar := archsimd.BroadcastUint16x8(uint16(r[1]))
	aq := archsimd.BroadcastInt16x8(q[1])
	ad := archsimd.BroadcastInt16x8(d[1])
	rr, qq, dd := ar.SetElem(0, uint16(r[0])), aq.SetElem(0, q[0]), ad.SetElem(0, d[0])
	max16 := archsimd.BroadcastUint16x8(32767)
	rate, err := archsimd.Uint32x4{}, archsimd.Uint32x4{}

	for i := 0; i < 128; i += 16 {
		h0 := archsimd.LoadInt16x8Array((*[8]int16)(unsafe.Add(cp, i)))
		h1 := archsimd.LoadInt16x8Array((*[8]int16)(unsafe.Add(cp, i+128)))
		h2 := archsimd.LoadInt16x8Array((*[8]int16)(unsafe.Add(cp, i+256)))
		h3 := archsimd.LoadInt16x8Array((*[8]int16)(unsafe.Add(cp, i+384)))
		u, v, w, z := h0.Add(h1).ShiftAllRight(1), h0.Sub(h1).ShiftAllRight(1), h2.Add(h3).ShiftAllRight(1), h2.Sub(h3).ShiftAllRight(1)
		o0, o1, o2, o3 := u.Add(w), v.Add(z), u.Sub(w), v.Sub(z)
		{
			mag := o0.Abs().ToBits()
			a := mag.Add(rr).Min(max16).BitsToInt16()
			t := blockYrdMulHigh16(a, qq)
			rate = rate.Add(blockYrdSumPairs16(t))
			diff := mag.Sub(t.Mul(dd).ToBits()).BitsToInt16()
			err = err.Add(blockYrdSquarePairs16(diff))
		}
		{
			mag := o1.Abs().ToBits()
			a := mag.Add(ar).Min(max16).BitsToInt16()
			t := blockYrdMulHigh16(a, aq)
			rate = rate.Add(blockYrdSumPairs16(t))
			diff := mag.Sub(t.Mul(ad).ToBits()).BitsToInt16()
			err = err.Add(blockYrdSquarePairs16(diff))
		}
		{
			mag := o2.Abs().ToBits()
			a := mag.Add(ar).Min(max16).BitsToInt16()
			t := blockYrdMulHigh16(a, aq)
			rate = rate.Add(blockYrdSumPairs16(t))
			diff := mag.Sub(t.Mul(ad).ToBits()).BitsToInt16()
			err = err.Add(blockYrdSquarePairs16(diff))
		}
		{
			mag := o3.Abs().ToBits()
			a := mag.Add(ar).Min(max16).BitsToInt16()
			t := blockYrdMulHigh16(a, aq)
			rate = rate.Add(blockYrdSumPairs16(t))
			diff := mag.Sub(t.Mul(ad).ToBits()).BitsToInt16()
			err = err.Add(blockYrdSquarePairs16(diff))
		}
		rr, qq, dd = ar, aq, ad
	}
	return int(blockYrdSum32(rate)), blockYrdSum32(err)
}
